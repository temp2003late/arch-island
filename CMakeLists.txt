cmake_minimum_required(VERSION 3.21)
project(ArchIsland VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.4 REQUIRED COMPONENTS Core Gui Qml Quick QuickControls2 Svg Concurrent ShaderTools)
add_library(island-core STATIC src/metrics.cpp src/metrics.h src/settings.cpp src/settings.h src/processmodel.cpp src/processmodel.h src/monitor.cpp src/monitor.h src/history.h)
target_include_directories(island-core PUBLIC src)
target_link_libraries(island-core PUBLIC Qt6::Core Qt6::Concurrent)
target_compile_options(island-core PRIVATE -Wall -Wextra -Wpedantic)
add_library(island-visuals STATIC src/charts.cpp src/charts.h)
target_include_directories(island-visuals PUBLIC src)
target_link_libraries(island-visuals PUBLIC Qt6::Quick Qt6::Qml)
target_compile_options(island-visuals PRIVATE -Wall -Wextra -Wpedantic)
file(GLOB ISLAND_ART CONFIGURE_DEPENDS assets/art/*.png assets/icons/*.svg)
set(ISLAND_RESOURCES qml/Main.qml qml/Island.qml qml/MetricCard.qml qml/IconButton.qml qml/ActionButton.qml assets/arch-island.svg assets/beam.svg assets/glow.svg assets/smoke.svg assets/cargo.svg assets/wake.svg ${ISLAND_ART})
qt_add_executable(arch-island src/main.cpp)
target_link_libraries(arch-island PRIVATE island-core island-visuals Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
qt_add_resources(arch-island visuals PREFIX "/" BASE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${ISLAND_RESOURCES})
qt_add_shaders(arch-island living_scene PREFIX "/" FILES assets/shaders/living.frag)
install(TARGETS arch-island RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/arch-island.desktop DESTINATION ${CMAKE_INSTALL_DATADIR}/applications)
install(FILES assets/arch-island.svg DESTINATION ${CMAKE_INSTALL_DATADIR}/icons/hicolor/scalable/apps)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/arch-island)
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test)
    qt_add_executable(island-tests tests/test_core.cpp)
    target_link_libraries(island-tests PRIVATE island-core Qt6::Test)
    add_test(NAME core COMMAND island-tests)
    qt_add_executable(island-gui-tests tests/test_gui.cpp)
    target_link_libraries(island-gui-tests PRIVATE island-core island-visuals Qt6::Test Qt6::Gui Qt6::Qml Qt6::Quick Qt6::QuickControls2 Qt6::Svg)
    qt_add_resources(island-gui-tests gui_visuals PREFIX "/" BASE ${CMAKE_CURRENT_SOURCE_DIR} FILES ${ISLAND_RESOURCES})
    qt_add_shaders(island-gui-tests living_scene_test PREFIX "/" FILES assets/shaders/living.frag)
    add_test(NAME gui-offscreen COMMAND island-gui-tests)
    set_tests_properties(gui-offscreen PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software" TIMEOUT 40)
endif()
